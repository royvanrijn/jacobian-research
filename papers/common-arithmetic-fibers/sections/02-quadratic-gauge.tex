\section{The universal quadratic gauge}\label{sec:gauge}

Let
\begin{equation}
 G(S)=g_1S+g_2S^2+\cdots+g_NS^N\in K[S],
 \qquad g_1g_3g_N\neq0.
 \label{eq:seed}
\end{equation}
In source coordinates $(x,y,z)$ put
\begin{equation}
 t=1+xy,
 \qquad
 q=t^2z+\frac{g_1}{g_3}y^2(1+3t).
 \label{eq:tq}
\end{equation}
Define $F_G=(\Pi,B,C):\A^3_K\to\A^3_K$ by
\begin{align}
 \Pi={}&tq,\notag\\
 B={}&y+3\frac{g_3}{g_1}xq+2\frac{g_2}{g_1}tq
       +\sum_{k=4}^N k\frac{g_k}{g_1}
          t^2x^{k-2}q^k,\notag\\
 C={}&x(5-3t)-\frac{g_3}{g_1}x^3z
       -\sum_{k=4}^N(k-2)\frac{g_k}{g_1}(xq)^k.
 \label{eq:gauge-map}
\end{align}
The formula is polynomial and uses only the coefficients of $G$.

For a target $(\Pi,B,C)$ define
\begin{equation}
 G_\Pi(S)=g_1S+\Pi(g_2S^2+g_3S^3)
          +\sum_{k=4}^Ng_k\Pi^kS^k
 \label{eq:Gpi}
\end{equation}
and
\begin{equation}
 E_{\Pi,B,C}(S)=G_\Pi(S)-\frac{g_1}{2}(BS^2+C).
 \label{eq:inverse}
\end{equation}
For $(\pi,b,c)\in K^*\times K^2$, let $\mathcal F_{\pi,b,c}$ denote the
fiber functor
\[
 \mathcal F_{\pi,b,c}(A)
 =\{(x,y,z)\in A^3:F_G(x,y,z)=(\pi,b,c)\}
\]
on commutative $K$-algebras.

\begin{lemma}[Derivative units over test algebras]\label{lem:derivative-unit}
Let $E\in K[S]$ be separable, let $A$ be a commutative $K$-algebra, and let
$s\in A$ satisfy $E(s)=0$.  Then $E'(s)$ is a unit of $A$.
\end{lemma}

\begin{proof}
Choose $U,V\in K[S]$ with $UE+VE'=1$.  Evaluating at $s$ gives
$V(s)E'(s)=1$.  Since $A$ is commutative, $V(s)$ is the inverse of $E'(s)$.
\end{proof}

\begin{proposition}[Quadratic-gauge fiber theorem]\label{prop:gauge}
The map $F_G$ satisfies
\[
 \det DF_G=-2.
\]
Its generic inverse equation is $E_{\Pi,B,C}(S)=0$.  Let
$(\pi,b,c)\in K^*\times K^2$.  If $E=E_{\pi,b,c}$ is squarefree of degree
$N$, then for every commutative $K$-algebra $A$ there is a natural bijection
\begin{equation}
 \operatorname{Hom}_{K\text{-alg}}(K[S]/(E),A)
 \simeq \mathcal F_{\pi,b,c}(A).
 \label{eq:natural-fiber-equivalence}
\end{equation}
Consequently, scheme-theoretically,
\[
 F_G^{-1}(\pi,b,c)\simeq\Spec K[S]/(E).
\]
In particular, this is a Keller fiber of rank $N$.
\end{proposition}

\begin{proof}
On $t\neq0$ introduce
\[
 S=\frac{x}{t},
 \qquad Q=y+xq,
 \qquad D=1-SQ+\Pi S^2=t^{-1}.
\]
A direct differentiation gives
\begin{equation}
 \det\frac{\partial(\Pi,S,Q)}{\partial(x,y,z)}=D^{-1}.
 \label{eq:source-chart-jacobian}
\end{equation}
Now set
\begin{equation}
 \beta(\Pi,S)
 =\frac{G_\Pi'(S)/g_1-1-\Pi S^2}{S}.
 \label{eq:beta}
\end{equation}
The numerator is divisible by $S$.  Substitution in
\eqref{eq:gauge-map} gives the two polynomial identities
\begin{equation}
 B=Q+\beta(\Pi,S),
 \qquad
 C=\frac{2G_\Pi(S)}{g_1}-BS^2.
 \label{eq:marked-line-identities}
\end{equation}
The second identity is exactly $E_{\Pi,B,C}(S)=0$, while differentiation
gives
\begin{equation}
 D=\frac{G_\Pi'(S)}{g_1}-BS
  =\frac{E_{\Pi,B,C}'(S)}{g_1}.
 \label{eq:D-derivative}
\end{equation}
At fixed $\Pi$, differentiating \eqref{eq:marked-line-identities} with
respect to $(S,Q)$ gives
\begin{equation}
 \det\frac{\partial(B,C)}{\partial(S,Q)}=-2D.
 \label{eq:plane-jacobian}
\end{equation}
Combining \eqref{eq:source-chart-jacobian} and
\eqref{eq:plane-jacobian} proves $\det DF_G=-2$ on the dense chart
$t\neq0$, hence polynomially everywhere.

It remains to identify the full fiber.  Fix $(\pi,b,c)$ and abbreviate
$E=E_{\pi,b,c}$.  Let $A$ be an arbitrary commutative $K$-algebra, and
denote the images of $\pi,b,c$ in $A$ by the same letters.  An $A$-point
$(x,y,z)$ of the fiber satisfies
\[
 tq=\pi.
\]
Since $\pi$ is a unit, both $t$ and $q$ are units in $A$.  Thus
\[
 S=xt^{-1},
 \qquad Q=y+xq,
 \qquad D=t^{-1}
\]
are defined globally on the whole fiber, not merely on its field-valued
points.  The polynomial identities \eqref{eq:marked-line-identities} and
\eqref{eq:D-derivative} give
\[
 Q+\beta(\pi,S)=b,
 \qquad E(S)=0,
 \qquad D=\frac{E'(S)}{g_1}.
\]
Hence every fiber point determines a root of $E$ in $A$.

Conversely, let $s\in A$ satisfy $E(s)=0$.  In characteristic zero,
squarefreeness implies separability, so Lemma~\ref{lem:derivative-unit} makes
$E'(s)$ a unit.  Therefore
\[
 d=\frac{E'(s)}{g_1}
\]
is a unit as well.  Put
\[
 Q=b-\beta(\pi,s),
 \qquad
 t=d^{-1},
 \qquad
 x=sd^{-1},
 \qquad
 y=Q-\pi s,
\]
and then
\[
 q=\pi d,
 \qquad
 z=d^2\left(q-\frac{g_1}{g_3}y^2(1+3t)\right).
\]
The derivative identity is precisely
\[
 d=1-sQ+\pi s^2.
\]
A direct ring calculation, valid in every commutative ring, now gives
\[
 1+xy=t,
 \qquad
 t^2z+\frac{g_1}{g_3}y^2(1+3t)=q,
 \qquad tq=\pi.
\]
The first marked identity gives $B=b$.  Since $E(s)=0$, the second gives
$C=c$.  Thus these formulas reconstruct an $A$-point of the fiber.

The two constructions are inverse.  Starting from a source point one recovers
\[
 D=t^{-1},\qquad x=S/D,\qquad y=Q-\pi S,\qquad q=\pi D,
\]
and the defining equation for $q$ recovers $z$.  Starting from a root, the
recovered marked coordinate is again $s$.

Finally, every displayed formula uses only ring operations, algebra maps, and
inversion of an explicitly supplied unit.  If $f:A\to A'$ is a morphism of
commutative $K$-algebras, then $f(d^{-1})=f(d)^{-1}$, so both constructions
commute with $f$.  The universal property of the quotient identifies roots of
$E$ in $A$ with $K$-algebra maps $K[S]/(E)\to A$ and turns the preceding
bijection into the natural equivalence \eqref{eq:natural-fiber-equivalence}.
Yoneda's lemma gives the asserted scheme isomorphism.  In particular, no
localization, nilpotent structure, or additional component is hidden by the
reconstruction.

The generic polynomial \eqref{eq:inverse} has degree $N$.  To see that it is
separable over $K(\Pi,B,C)$, write
\[
 E_{\Pi,B,C}(S)=H_{\Pi,B}(S)-\frac{g_1}{2}C.
\]
Its derivative is $H_{\Pi,B}'(S)$ and has degree $N-1$.  The resultant
\[
 \operatorname{Res}_S(E_{\Pi,B,C},E_{\Pi,B,C}')
\]
is a polynomial in the transcendental parameter $C$ of degree $N-1$.  Up to
sign, its leading coefficient is
\[
 \operatorname{lc}(E_{\Pi,B,C}')^N
 \left(\frac{g_1}{2}\right)^{N-1}.
\]
This is nonzero: the derivative has leading coefficient $3g_3\Pi$ when
$N=3$, and $Ng_N\Pi^N$ when $N\geq4$.  Hence the resultant is nonzero, so the
generic inverse polynomial is separable.  The same reconstruction works at
every generic root, and therefore $\gdeg(F_G)=N$.
\end{proof}

\begin{remark}[Logical separation of the hypotheses]\label{rem:gauge-hypotheses}
The represented-fiber argument has a smaller logical core than the displayed
map.  It uses that $\pi$ is a unit, that $E$ is separable, and that
\[
 E'(S)=g_1\bigl(1-bS+S\beta(\pi,S)+\pi S^2\bigr).
\]
Once the source coefficient $g_1/g_3$ is supplied, the root/source
reconstruction itself does not use a further polynomial identity.  The
condition $g_3\neq0$ is needed to obtain that coefficient from the explicit
quadratic gauge.  For a translated polynomial $P(a+S)-P(a)$, the natural
cubic coefficient is the third Hasse derivative at $a$; in characteristic
zero it is $P'''(a)/3!$.  This separation is also the organization used by
the Lean formalization.
\end{remark}

\begin{corollary}[Jacobian-one effective form]\label{cor:effective-gauge}
Let
\[
 L(\Pi,B,C)=\left(\Pi,-\frac{B}{2},C\right),
 \qquad \widetilde F_G=L\circ F_G.
\]
Then
\[
 \det D\widetilde F_G=1,
 \qquad \gdeg(\widetilde F_G)=N.
\]
For every $(\pi,b,c)\in K^*\times K^2$ for which $E_{\pi,b,c}$ is
squarefree of degree $N$,
\[
 \widetilde F_G^{-1}\left(\pi,-\frac b2,c\right)
 \simeq\Spec K[S]/(E_{\pi,b,c}).
\]
In particular, every target with $b=0$ is fixed by the normalization.
Moreover, in total degree,
\[
 \deg\Pi\leq7,
 \qquad \deg B\leq6N+2,
 \qquad \deg C\leq6N,
\]
and therefore
\[
 \boxed{\max_i\deg(\widetilde F_G)_i\leq6N+2.}
\]
\end{corollary}

\begin{proof}
The map $L$ is a linear automorphism of determinant $-1/2$.  The determinant,
geometric-degree, and fiber statements therefore follow from
Proposition~\ref{prop:gauge}.

For the degree estimate, $\deg t=2$ and $\deg q\leq5$, hence
$\deg\Pi\leq7$.  The summand of $B$ indexed by $k\geq4$ has degree at most
\[
 2\deg t+(k-2)+k\deg q\leq4+(k-2)+5k=6k+2,
\]
while the three terms preceding the sum have degrees at most $1$, $6$, and
$7$.  Thus $\deg B\leq6N+2$.  Similarly, the summand $(xq)^k$ in $C$ has
degree at most $6k$, while the preceding terms have degrees at most $3$ and
$4$, so $\deg C\leq6N$.  Linear output normalization does not change these
bounds.
\end{proof}
