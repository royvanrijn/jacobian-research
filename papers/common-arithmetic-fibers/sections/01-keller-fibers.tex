\section{Prescribing finite \'etale algebras}
\label{sec:prescribing}

The construction used in the proof can be read from left to right:
\[
 \boxed{
 \text{squarefree }P(T)
 \ \longmapsto\
 G(S)=P(a+S)-P(a)
 \ \longmapsto\
 \widetilde F_G^{-1}(y_{P,a})\simeq\Spec K[T]/(P).}
\]
The last arrow is an isomorphism of schemes, not merely a matching of
\(K\)-points or a count after passing to an algebraic closure.

\begin{theorem}[Polynomial realization]\label{thm:polynomial-realization}
Let \(P\in K[T]\) be squarefree of degree \(N\geq3\), and let
\(a\in K\) satisfy
\[
 P'(a)P'''(a)\neq0.
\]
Put \(G(S)=P(a+S)-P(a)\), let \(F_G=(\Pi,B,C)\) be the
quadratic-gauge map of \eqref{eq:gauge-map}, and set
\[
 \widetilde F_G=L\circ F_G,
 \qquad
 L(\Pi,B,C)=\left(\Pi,-\frac B2,C\right).
\]
Then
\[
 \det D\widetilde F_G=1,
 \qquad
 \gdeg(\widetilde F_G)=N.
\]
At
\begin{equation}
 y_{P,a}=\left(1,0,-\frac{2P(a)}{P'(a)}\right)
 \label{eq:main-target}
\end{equation}
the full Keller fiber is
\[
 \boxed{\widetilde F_G^{-1}(y_{P,a})
 \simeq\Spec K[T]/(P).}
\]
\end{theorem}

\begin{proof}
The linear and cubic coefficients of \(G\) are
\(P'(a)\) and \(P'''(a)/3!\), so the quadratic gauge of
Section~\ref{sec:gauge} is defined.  At the target
\eqref{eq:main-target}, its inverse polynomial is
\[
 G(S)-\frac{P'(a)}2\left(-\frac{2P(a)}{P'(a)}\right)
 =P(a+S).
\]
Translation preserves squarefreeness, so
Corollary~\ref{cor:squarefree-gauge} identifies the fiber with
\(\Spec K[S]/(P(a+S))\).

Translation back to the original polynomial gives
\[
 K[S]/(P(a+S))\longrightarrow K[T]/(P(T)),
 \qquad \overline S\longmapsto\overline T-a,
\]
with inverse \(\overline T\mapsto\overline S+a\).  The output map \(L\)
fixes the chosen target, multiplies the Jacobian by \(-1/2\), and changes
neither geometric degree nor fiber length.  Corollary~\ref{cor:effective-gauge}
therefore gives \(\det D\widetilde F_G=1\) and
\(\gdeg(\widetilde F_G)=N\).  Since the displayed fiber has length \(N\),
it is full.
\end{proof}

\begin{corollary}[Effective and functorial form]
\label{cor:polynomial-effective}
Under the hypotheses of Theorem~\ref{thm:polynomial-realization},
\[
 \max_i\deg(\widetilde F_G)_i\leq6N+2.
\]
For every commutative \(K\)-algebra \(R\), the scheme isomorphism induces a
bijection
\[
 \operatorname{Hom}_{K\text{-alg}}(K[T]/(P),R)
 \simeq
 \{u\in R^3:\widetilde F_G(u)=y_{P,a}\},
\]
natural in \(R\).
\end{corollary}

\begin{proof}
The degree bound is Corollary~\ref{cor:effective-gauge}.  The natural
bijection is the represented-fiber equivalence in
Corollary~\ref{cor:squarefree-gauge}, composed with the translation
\(K[S]/(P(a+S))\simeq K[T]/(P)\).
\end{proof}

\begin{lemma}[Monogenicity]\label{lem:monogenic}
Every finite \'etale algebra over an infinite field is generated by one
element.  Equivalently, if \(A\) has rank \(N\), then
\[
 A\simeq K[T]/(P)
\]
for a squarefree polynomial \(P\in K[T]\) of degree \(N\).
\end{lemma}

\begin{proof}
For \(\theta\in A\), let \(\chi_\theta(T)\) be the characteristic polynomial
of multiplication by \(\theta\) on the \(K\)-vector space \(A\), and set
\[
 \Delta(\theta)=\operatorname{disc}(\chi_\theta).
\]
This is a polynomial function on the affine space underlying \(A\).  After
extending scalars to a separable closure, identify
\(A\otimes_KK^{\rm sep}\) with \((K^{\rm sep})^N\).  In these coordinates,
\[
 \Delta(\theta_1,\ldots,\theta_N)
 =\prod_{i<j}(\theta_i-\theta_j)^2,
\]
so \(\Delta\) is not the zero polynomial.  Since \(K\) is infinite, some
\(\theta\in A\) satisfies \(\Delta(\theta)\neq0\).  Its characteristic
polynomial has \(N\) distinct geometric roots, and its minimal polynomial
therefore has degree \(N\).  Hence \(K[\theta]=A\), with squarefree minimal
polynomial \(P\).
\end{proof}

\begin{corollary}[Finite \'etale realization]
\label{cor:finite-etale-realization}
Every finite \'etale \(K\)-algebra \(A\) of rank \(N\geq3\) occurs as a full
Keller fiber of a Jacobian-one map
\[
 \A^3_K\longrightarrow\A^3_K
\]
of geometric degree \(N\).

More precisely, once an isomorphism \(A\simeq K[T]/(P)\) and an element
\(a\in K\) with \(P'(a)P'''(a)\neq0\) are supplied, the map, target, and
fiber isomorphism are given by the explicit formulas of
Theorem~\ref{thm:polynomial-realization}.  For an abstract \(A\), the
monogenicity argument proves that such a presentation exists; it does not
choose one canonically.
\end{corollary}

\begin{proof}
Lemma~\ref{lem:monogenic} supplies a squarefree presentation
\(A\simeq K[T]/(P)\) of degree \(N\).  The polynomials \(P'\) and \(P'''\)
are nonzero in characteristic zero, and \(K\) is infinite, so some \(a\)
avoids the finite zero set of \(P'P'''\).  Apply
Theorem~\ref{thm:polynomial-realization}.
\end{proof}

\begin{corollary}[Rank classification, using the classical Galois case]
\label{cor:rank-classification}
Let \(X\) be a finite \(K\)-scheme of positive rank \(N\).  Then \(X\) is
isomorphic to a full Keller fiber of a geometric-degree-\(N\) Keller map
\[
 F:\A^3_K\longrightarrow\A^3_K
\]
if and only if \(X\) is finite \'etale and \(N\neq2\).  Consequently, the
possible nonzero ranks of full Keller fibers in \(\A^3_K\) are exactly
\[
 \boxed{1,3,4,5,\ldots}.
\]
\end{corollary}

\begin{proof}
Every Keller fiber is finite \'etale.  If it is a full Keller fiber of rank
\(N\), its map has geometric degree \(N\).  A degree-two function-field
extension in characteristic zero is Galois, so the classical Galois case of
the Jacobian theorem excludes \(N=2\)
\cite{Campbell,Razar,Wright}.  For completeness, the usual complex statement
applies over an arbitrary characteristic-zero field by spreading out the
finitely many coefficients, observing that generic degree is preserved by
scalar extension, embedding the coefficient field in \(\mathbb C\), and
descending the polynomial inverse by faithful flatness.  Conversely, the identity
realizes every rank-one finite \'etale \(K\)-scheme, namely \(\Spec K\), and
Corollary~\ref{cor:finite-etale-realization} realizes every finite \'etale
scheme of rank \(N\geq3\) as a full Keller fiber.  Empty and shorter special
fibers are Keller fibers, but they are not full Keller fibers and hence do
not alter the rank list.
\end{proof}
