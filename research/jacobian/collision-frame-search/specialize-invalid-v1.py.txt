#!/usr/bin/env python3
"""Exact specializations of the complete critical projection for boundary shears."""
from pathlib import Path
import gzip,hashlib,json,time
import sympy as sp
HERE=Path(__file__).resolve().parent
w,t,a=sp.symbols("w t a")
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,v):
    raw=(json.dumps(v,indent=2,sort_keys=True)+"\n").encode()
    data=gzip.compress(raw,mtime=0) if p.suffix==".gz" else raw
    temp=p.with_suffix(p.suffix+".research/.scratch");temp.write_bytes(data);temp.replace(p)
def key_bivariate(expr):return str(sp.Poly(expr,w,t,domain=sp.QQ).monic().as_expr())
def gcd_basis(polynomials):
    basis=[]
    for p in polynomials:
        if p.degree()<=0:continue
        squarefree=p.exquo(p.gcd(p.diff(p.ring.gens[0]))).monic()
        if squarefree not in basis:basis.append(squarefree)
    while True:
        changed=False
        for i in range(len(basis)):
            for j in range(i):
                f,g=basis[i],basis[j];h=f.gcd(g)
                if h.degree()>0:
                    rest=[b for k,b in enumerate(basis) if k not in (i,j)]
                    for q in (h.monic(),f.exquo(h).monic(),g.exquo(h).monic()):
                        if q.degree()>0 and q not in rest:rest.append(q)
                    basis=rest;changed=True;break
            if changed:break
        if not changed:return basis
def serialize(poly):
    return [dict(t_degree=e[0],w_coefficients=list(map(str,reversed(c.to_list()))))
            for e,c in sorted(poly.items())]

def main():
    family=json.loads((HERE/"two-parameter-family.json").read_text())
    projection=json.loads((HERE/"projection-factors.json").read_text())
    sources=json.loads((HERE/"projection-checkpoint.json").read_text())
    assert projection["status"]==sources["status"]=="complete"
    curves=family["factors"]
    by_key={key_bivariate(row["polynomial"]):i for i,row in enumerate(curves)}
    moving=[i for i,row in enumerate(curves) if row["degree_t"]>0]
    vertical=[i for i,row in enumerate(curves) if row["degree_t"]==0]
    # A potential common root is possible only along a projected resultant.
    links={row["polynomial"]:[] for row in projection["factors"]}
    for row in sources["resultants"]:
        if row["source"][0]!="pair":continue
        left=by_key[key_bivariate(row["source"][1])]
        right=by_key[key_bivariate(row["source"][2])]
        for f,m in sp.factor_list(sp.Poly(row["polynomial"],w,domain=sp.QQ))[1]:
            links[str(f.monic().as_expr())].append((left,right))
    prepared={}
    for i in moving:
        p=sp.Poly(curves[i]["polynomial"],t)
        prepared[i]=[(e[0],sp.Poly(c,w,domain=sp.QQ).all_coeffs()) for e,c in p.terms()]
    out=HERE/"specializations";out.mkdir(exist_ok=True)
    summaries=[];minimum=family["generic_support"]
    started=time.monotonic()
    for number,row in enumerate(projection["factors"]):
        path=out/f"case-{number:04d}.json.gz"
        if path.exists():
            with gzip.open(path,"rt") as f:result=json.load(f)
            assert result["source_sha256"]==sha(Path(__file__))
            assert result["projection_polynomial"]==row["polynomial"]
        else:
            r=sp.Poly(row["polynomial"],w,domain=sp.QQ)
            assert r.is_irreducible
            K=sp.QQ.algebraic_field(sp.AlgebraicNumber((r,a)))
            ring=K.poly_ring(t).ring
            polys={}
            for i in moving:
                terms={}
                for degree,coefficients in prepared[i]:
                    value=K.zero
                    for c in coefficients:value=value*K.unit+K.convert(c)
                    if value:terms[(degree,)]=value
                polys[i]=ring.from_dict(terms)
                assert polys[i],"Primitive irreducible moving curve cannot vanish identically"
            always=set()
            for i in vertical:
                if sp.Poly(curves[i]["polynomial"],w,domain=sp.QQ).rem(r).is_zero:
                    always.update(curves[i]["vanishing_coefficients"])
            parent={i:i for i in moving}
            def find(i):
                while parent[i]!=i:
                    parent[i]=parent[parent[i]];i=parent[i]
                return i
            for left,right in links[row["polynomial"]]:parent[find(left)]=find(right)
            groups={}
            for i in moving:groups.setdefault(find(i),[]).append(i)
            bases=[];local_minimum=family["generic_support"]-len(always)
            for group in groups.values():
                roots=gcd_basis([polys[i] for i in group])
                for f in roots:
                    zeros=set(always)
                    containing=[]
                    for i in group:
                        if not polys[i].rem(f):
                            containing.append(i)
                            zeros.update(curves[i]["vanishing_coefficients"])
                    support=family["generic_support"]-len(zeros)
                    local_minimum=min(local_minimum,support)
                    bases.append(dict(group=group,polynomial=serialize(f),
                                      containing_curves=containing,zero_coefficients=sorted(zeros),
                                      support=support))
                for i in group:
                    residual=polys[i]
                    for f in roots:
                        while residual.degree()>=f.degree() and not residual.rem(f):
                            residual=residual.exquo(f)
                    assert residual.degree()==0
            result=dict(status="complete",source_sha256=sha(Path(__file__)),
                        family_sha256=sha(HERE/"two-parameter-family.json"),
                        projection_polynomial=row["polynomial"],projection_degree=r.degree(),
                        always_zero=sorted(always),generic_specialized_support=family["generic_support"]-len(always),
                        minimum=local_minimum,bases=bases,potential_links=links[row["polynomial"]])
            save(path,result)
        summaries.append(dict(path=str(path.relative_to(HERE)),sha256=sha(path),
                              projection_polynomial=row["polynomial"],minimum=result["minimum"]))
        minimum=min(minimum,result["minimum"])
        if number%10==0 or result["minimum"]<48:
            print(json.dumps(dict(done=number+1,total=len(projection["factors"]),
                                  local=result["minimum"],minimum=minimum,
                                  polynomial=row["polynomial"],elapsed=time.monotonic()-started)),flush=True)
        save(HERE/"specialization-progress.json",dict(status="running",done=number+1,
             total=len(projection["factors"]),minimum=minimum,source_sha256=sha(Path(__file__))))
    # Away from the projection roots, only one moving factor can vanish.
    generic_minimum=family["generic_support"]-max(len(curves[i]["vanishing_coefficients"]) for i in moving)
    minimum=min(minimum,generic_minimum)
    manifest=dict(status="complete",source_sha256=sha(Path(__file__)),
                  family_sha256=sha(HERE/"two-parameter-family.json"),
                  projection_sha256=sha(HERE/"projection-factors.json"),
                  cases=summaries,case_count=len(summaries),
                  generic_minimum=generic_minimum,minimum=minimum,
                  elapsed_seconds=time.monotonic()-started)
    save(HERE/"specialization-manifest.json",manifest)
    print(json.dumps({k:v for k,v in manifest.items() if k!="cases"}),flush=True)
if __name__=="__main__":main()
